Daily drivers
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
List<String> line = new ArrayList<>();
line.add("Ada");
line.add("Bob");
Map<String, String> locker = new HashMap<>();
locker.put("A1", "Ada bag");
// line.get(0)="Ada"  locker.get("A1")="Ada bag"
// line.size()=2  locker.containsKey("A1")=true
Lists: brackets are illegal — get(0)! Maps: absent keys give null — unbox blindly, detonate.
