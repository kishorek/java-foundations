Step 2

% keeps leftovers

7 % 3 = 1. Evenness test inside.

ProgramRead each line in order
public class Main {
  public static void main(String[] args) {
    int age = 20;
    boolean teen = age >= 13 && age <= 19;
    System.out.println(teen); // false: 20 fails the second half
    System.out.println(7 % 3); // 1: the leftover
    int x = 5;
    x += 2; // x = x + 2
    System.out.println(x); // 7
    int i = 5;
    System.out.println(i++); // 5: use-then-step
    System.out.println(i);   // 6: stepped now
  }
}
